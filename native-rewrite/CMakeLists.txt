cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" bridge_version LIMIT_COUNT 1)
project(DoubaoVRNative VERSION ${bridge_version} LANGUAGES CXX RC)
configure_file(core/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The native bridge requires Windows x64.")
endif()
unset(POLICY_POWERSHELL CACHE)
find_program(POLICY_POWERSHELL pwsh.exe REQUIRED)
execute_process(COMMAND "${POLICY_POWERSHELL}" -NoProfile -File "${CMAKE_SOURCE_DIR}/scripts/check-policy.ps1"
    COMMAND_ERROR_IS_FATAL ANY)
add_custom_target(enforce_policy ALL
    COMMAND "${POLICY_POWERSHELL}" -NoProfile -File "${CMAKE_SOURCE_DIR}/scripts/check-policy.ps1")
set(openvr_SOURCE_DIR "${CMAKE_SOURCE_DIR}/dependencies/openvr")
file(READ "${CMAKE_SOURCE_DIR}/dependencies.lock.json" dependency_lock)
foreach(index RANGE 0 3)
    string(JSON sdk_path GET "${dependency_lock}" sdkFiles ${index} path)
    string(JSON sdk_hash GET "${dependency_lock}" sdkFiles ${index} sha256)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/dependencies/${sdk_path}")
        message(FATAL_ERROR "Missing SDK file ${sdk_path}. Run build.ps1 to prepare the verified SDK.")
    endif()
    file(SHA256 "${CMAKE_SOURCE_DIR}/dependencies/${sdk_path}" actual_hash)
    if(NOT actual_hash STREQUAL sdk_hash)
        message(FATAL_ERROR "SDK checksum mismatch: ${sdk_path}")
    endif()
endforeach()
add_library(DoubaoVR.Core SHARED core/bridge.cpp core/graphics.cpp core/platform.cpp core/audio.cpp core/microphone_keeper.cpp core/caption_close.cpp "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
add_dependencies(DoubaoVR.Core enforce_policy)
target_compile_features(DoubaoVR.Core PRIVATE cxx_std_20)
target_compile_definitions(DoubaoVR.Core PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN BRIDGE_EXPORTS)
target_compile_options(DoubaoVR.Core PRIVATE /W4 /WX /EHsc /utf-8 /permissive- /sdl)
target_include_directories(DoubaoVR.Core PRIVATE core "${openvr_SOURCE_DIR}/headers")
target_link_libraries(DoubaoVR.Core PRIVATE d3d11 dxgi d3dcompiler windowsapp ole32
    "${openvr_SOURCE_DIR}/lib/win64/openvr_api.lib")
set_property(TARGET DoubaoVR.Core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(DoubaoVR.Core PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Release" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
add_custom_command(TARGET DoubaoVR.Core POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${openvr_SOURCE_DIR}/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:DoubaoVR.Core>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${openvr_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:DoubaoVR.Core>/OpenVR-LICENSE.txt")
option(BUILD_TESTING "Build local developer checks" OFF)
if(BUILD_TESTING)
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/tests/CMakeLists.txt")
        message(FATAL_ERROR "Local validation sources are required when BUILD_TESTING is enabled.")
    endif()
    include(CTest)
    add_subdirectory(tests)
endif()
